python
from functools import wraps
from flask import jsonify
from flask_jwt_extended import get_jwt_identity
from models import User

def role_required(*allowed_roles):
    """角色权限装饰器"""
    def decorator(f):
        @wraps(f)
        def decorated_function(*args, **kwargs):
            user_id = get_jwt_identity()
            user = User.query.get(int(user_id))
            if not user:
                return jsonify({'error': '用户不存在'}), 404
            if user.role not in allowed_roles:
                return jsonify({'error': '权限不足'}), 403
            return f(*args, **kwargs)
        return decorated_function
    return decorator

def admin_required(f):
    """管理员权限"""
    return role_required('admin')(f)

def teacher_required(f):
    """教师及以上权限"""
    return role_required('admin', 'head', 'advisor', 'faculty', 'teaching_admin')(f)

def advisor_required(f):
    """班主任权限"""
    return role_required('admin', 'head', 'advisor')(f)

第八阶段：项目问题修复汇总
对话 20：字符集问题总结
问题： MySQL 插入中文时报错 Incorrect string value

尝试的解决方案：

SET NAMES utf8mb4 → 临时生效，下次连接失效

ALTER TABLE CONVERT TO CHARACTER SET utf8mb4 → 外键约束阻止

修改 my.ini → 权限不足/找不到文件

删除外键后修改 → 仍有个别字段报错

重建表 → 插入时仍报错

根本原因： MySQL 服务端 character_set_server 不是 utf8mb4，而修改配置文件需要管理员权限且文件位置不确定。

最终决定： 采用拼音存储，前端映射显示中文。
